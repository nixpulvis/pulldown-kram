10/// What a stand-in, put in the text by [`StandIns`], stands for.
11#[derive(Debug, Clone, Copy, PartialEq, Eq)]
12pub(crate) enum StandIn<'a> {
13    /// Nothing: an end-of-block marker, or an extension that adds nothing.
14    Nothing,
15    /// A `comment` extension's text.
16    Comment(&'a str),
17    /// A `nomarkdown` extension's text, to come out as raw HTML.
18    Raw(&'a str),
19}

21/// A stand-in in the text.
22#[derive(Debug, Clone, Copy, PartialEq, Eq)]
23pub(crate) struct Mark<'a> {
24    /// Where it starts.
25    pub at: usize,
26    /// Whether it stands in for a block, rather than a span.
27    pub block: bool,
28    pub stands_for: StandIn<'a>,
29}

31/// Syntax of kramdown's own that CommonMark reads as plain text, replaced
32/// with stand-ins the reader swaps back.
33///
34/// That covers the end-of-block marker, a line of only `^`, the `comment`
35/// and `nomarkdown` extensions, `{::comment}…{:/}`, and abbreviation
36/// definitions. A block's stand-in is an empty HTML comment, which ends the
37/// same blocks a marker does. A span's is an autolink, which can't start an
38/// HTML block. An extension tag that can't be read, like one never closed,
39/// is escaped, so it stays text as in kramdown.
40///
41/// The same pass also puts a header row in each table without one, which
42/// the reader leaves out again, since it needs to know where blocks start
43/// just as the stand-ins do.
44pub(crate) struct StandIns<'a> {
45    markdown: &'a str,
46    out: Rewrite,
47    marks: Vec<Mark<'a>>,
48    end_of_blocks: bool,
49    comments: bool,
50    nomarkdown: bool,
51    tables: bool,
52    /// Whether `$$` math is read. A `|` inside math doesn't end a cell.
53    math: bool,
54    /// The abbreviations defined, if they're read.
55    abbreviations: Option<Vec<Abbreviation<'a>>>,
56}

58/// Markdown with stand-ins in it, by [`StandIns`].
59pub(crate) struct Marked<'a> {
60    pub text: Cow<'a, str>,
61    pub offsets: Offsets,
62    pub marks: Vec<Mark<'a>>,
63    pub abbreviations: Vec<Abbreviation<'a>>,
64}

66/// Reads the lines of kramdown's tables.
67struct TableLine;
68
69impl TableLine {
70    /// Where each `|` in `line` is that isn't escaped, in code, or in math
71    /// when `math` is read.
72    fn pipes(line: &str, math: bool) -> Vec<usize> {
73        let code = Self::code_spans(line);
74        let math = if math {
75            Self::math_spans(line)
76        } else {
77            Vec::new()
78        };
79        Self::unescaped_pipes(line)
80            .filter(|at| !code.iter().chain(&math).any(|span| span.contains(at)))
81            .collect()
82    }
83
84    /// Where the math between `$$` in `line` is.
85    fn math_spans(line: &str) -> Vec<Range<usize>> {
86        let mut spans = Vec::new();
87        let mut from = 0;
88        while let Some(open) = line[from..].find("$$").map(|at| from + at + 2)
89            && let Some(close) = line[open..].find("$$").map(|at| open + at)
90        {
91            spans.push(open..close);
92            from = close + 2;
93        }
94        spans
95    }
96
97    /// Where each `|` in `line` is that's in code and not escaped.
98    fn code_pipes(line: &str) -> Vec<usize> {
99        let code = Self::code_spans(line);
100        Self::unescaped_pipes(line)
101            .filter(|at| code.iter().any(|span| span.contains(at)))
102            .collect()
103    }
104
105    fn unescaped_pipes(line: &str) -> impl Iterator<Item = usize> + '_ {
106        line.match_indices('|')
107            .map(|(at, _)| at)
108            .filter(move |&at| {
109                let backslashes = line[..at].len() - line[..at].trim_end_matches('\\').len();
110                backslashes.is_multiple_of(2)
111            })
112    }
113
114    /// Where the code spans closed in `line` are.
115    fn code_spans(line: &str) -> Vec<Range<usize>> {
116        let mut spans = Vec::new();
117        let mut from = 0;
118        while let Some(open) = line[from..].find('`').map(|at| from + at) {
119            let (run, close) = code_span(line, open);
120            match close {
121                Some(close) => {
122                    spans.push(open + run..close);
123                    from = close + run;
124                }
125                None => from = open + run,
126            }
127        }
128        spans
129    }
130
131    /// How many cells the row `line` has.
132    fn cells(line: &str, math: bool) -> usize {
133        let row = line.trim();
134        let pipes = Self::pipes(row, math);
135        let mut cells = pipes.len() + 1;
136        if row.starts_with('|') {
137            cells -= 1;
138        }
139        if row.len() > 1 && pipes.last() == Some(&(row.len() - 1)) {
140            cells -= 1;
141        }
142        cells.max(1)
143    }
144
145    /// Whether `line` separates parts of a table, like `|---|---|`.
146    fn separator(line: &str) -> bool {
147        let line = line.trim();
148        line.contains(['-', '='])
149            && line
150                .chars()
151                .all(|c| matches!(c, '|' | '+' | '-' | ':' | '=' | ' ' | '\t'))
152    }
153
154    /// The header separator `line`, written as CommonMark needs it, with a
155    /// cell for each of `columns`, aligned as in `line`. `None` if it's
156    /// fine as it is. kramdown's may have fewer cells, and `+` between them.
157    fn delimiter(line: &str, columns: usize) -> Option<String> {
158        let line = line.trim();
159        let inside = line.strip_prefix(['|', '+']).unwrap_or(line);
160        let inside = inside.strip_suffix(['|', '+']).unwrap_or(inside);
161        let cells: Vec<&str> = inside.split(['|', '+']).map(str::trim).collect();
162        if cells.len() == columns && !line.contains('+') {
163            return None;
164        }
165        let mut delimiter = String::from("|");
166        for n in 0..columns {
167            let cell = cells.get(n).copied().unwrap_or("-");
168            delimiter.push_str(match (cell.starts_with(':'), cell.ends_with(':')) {
169                (true, true) => ":-:",
170                (true, false) => ":-",
171                (false, true) => "-:",
172                (false, false) => "-",
173            });
174            delimiter.push('|');
175        }
176        Some(delimiter)
177    }
178}

180/// An extension's start tag, like `{::comment}` or `{::nomarkdown/}`.
181struct StartTag<'a> {
182    name: &'a str,
183    /// How long it is.
184    len: usize,
185    /// Whether it closes itself, holding nothing.
186    closed: bool,
187    /// Whether what it holds comes out as raw HTML, as a `nomarkdown`'s does
188    /// unless its `type` names another converter.
189    html: bool,
190}

192impl<'a> StandIns<'a> {
193    /// What stands in for a block.
194    pub const BLOCK: &'static str = "<!---->";
195    /// What stands in for a span.
196    pub const SPAN: &'static str = "<kd:x>";

198    /// `markdown` with its stand-ins and table headers put in, for the
199    /// syntax `extensions` reads.
200    ///
201    /// Along with the text and the offsets of its edits come the stand-ins,
202    /// in order, and the abbreviations defined.
203    pub fn rewrite(markdown: &'a str, extensions: Extensions) -> Marked<'a> {
204        let reads = |extension| extensions.contains(extension);
205        let markers = reads(Extensions::ENABLE_END_OF_BLOCK_MARKERS) && markdown.contains('^');
206        let tags = (reads(Extensions::ENABLE_COMMENTS) || reads(Extensions::ENABLE_NOMARKDOWN))
207            && markdown.contains("{:");
208        let tables = reads(Extensions::ENABLE_TABLES_WITHOUT_HEADERS)
209            && reads(Extensions::ENABLE_TABLES)
210            && markdown.contains('|');
211        let abbreviations = reads(Extensions::ENABLE_ABBREVIATIONS) && markdown.contains("*[");
212        if !markers && !tags && !tables && !abbreviations {
213            return Marked {
214                text: Cow::Borrowed(markdown),
215                offsets: Offsets::default(),
216                marks: Vec::new(),
217                abbreviations: Vec::new(),
218            };
219        }
220        let mut stand_ins = Self {
221            markdown,
222            out: Rewrite::of(markdown),
223            marks: Vec::new(),
224            abbreviations: abbreviations.then(Vec::new),
225            end_of_blocks: reads(Extensions::ENABLE_END_OF_BLOCK_MARKERS),
226            comments: reads(Extensions::ENABLE_COMMENTS),
227            nomarkdown: reads(Extensions::ENABLE_NOMARKDOWN),
228            tables,
229            math: reads(Extensions::ENABLE_MATH),
230        };
231        stand_ins.lines();
232        let (text, offsets) = stand_ins.out.finish(markdown);
233        Marked {
234            text,
235            offsets,
236            marks: stand_ins.marks,
237            abbreviations: stand_ins.abbreviations.unwrap_or_default(),
238        }
239    }

241    /// Replaces the abbreviation definition at the line `start` with a
242    /// stand-in, if there's one there, along with the lines of only
243    /// attribute lists right before and after it, which belong to it.
244    ///
245    /// Returns the index of the last line replaced. Its line break is left
246    /// for the caller to copy.
247    fn abbreviation(&mut self, lines: &[&'a str], start: usize) -> Option<usize> {
248        self.abbreviations.as_ref()?;
249        let only_lists = |line: &'a str| AttributeLists::whole(line.trim());
250        let mut attribute_lists = AttributeLists::default();
251        let mut end = start;
252        while let Some(before) = lines.get(end).copied().and_then(only_lists) {
253            attribute_lists.extend(before);
254            end += 1;
255        }
256        let mut abbreviation = Abbreviation::defined(lines.get(end)?)?;
257        while let Some(after) = lines.get(end + 1).copied().and_then(only_lists) {
258            attribute_lists.extend(after);
259            end += 1;
260        }
261        abbreviation.attribute_lists = attribute_lists;
262        self.abbreviations.as_mut()?.push(abbreviation);
263        let line = lines[start];
264        let lead = line.len() - line.trim_start().len();
265        let len = self.offset(lines[end]) + lines[end].trim_end().len() - self.offset(line) - lead;
266        self.stand_in(&line[..lead], len, true, StandIn::Nothing);
267        Some(end)
268    }

270    fn lines(&mut self) {
271        let lines: Vec<&'a str> = self.markdown.split_inclusive('\n').collect();
272        let mut fence = Fence::default();
273        // Whether the lines are in an HTML block, where nothing is kramdown.
274        let mut html = false;
275        let mut after_blank = true;
276        // Whether a block may start on the line, as a table only does.
277        let mut boundary = true;
278        let mut i = 0;
279        while i < lines.len() {
280            let line = lines[i];
281            i += 1;
282            let starts_block = std::mem::replace(&mut boundary, false);
283            if fence.read(line) {
284                self.out.keep(line);
285                after_blank = false;
286                continue;
287            }
288            let trimmed = line.trim();
289            let Lead {
290                len: lead,
291                indent,
292                quoted,
293            } = Lead::of(line);
294            if trimmed.is_empty() {
295                html = false;
296                after_blank = true;
297                boundary = true;
298                self.out.keep(line);
299                continue;
300            }
301            html |= after_blank && line[lead..].starts_with('<');
302            after_blank = false;
303            if html || indent >= 4 {
304                self.out.keep(line);
305                continue;
306            }
307            let content = line[lead..].trim_end();
308            boundary = content.starts_with('#') || AttributeLists::whole(content).is_some();
309            if starts_block
310                && !quoted
311                && let Some(end) = self.abbreviation(&lines, i - 1)
312            {
313                let last = lines[end];
314                self.out.keep(&last[last.trim_end().len()..]);
315                i = end + 1;
316                boundary = true;
317                continue;
318            }
319            if self.end_of_blocks && content == "^" {
320                self.stand_in(&line[..lead], content.len(), true, StandIn::Nothing);
321                self.out.keep(&line[lead + content.len()..]);
322                boundary = true;
323                continue;
324            }
325            if !quoted && let Some(end) = self.block(&lines, i - 1, lead) {
326                let last = lines[end];
327                let last_content = last.trim_end().len();
328                self.out.keep(&last[last_content..]);
329                i = end + 1;
330                continue;
331            }
332            if self.tables
333                && starts_block
334                && !quoted
335                && let Some(end) = self.table(&lines, i - 1)
336            {
337                i = end;
338                continue;
339            }
340            if self.comments || self.nomarkdown {
341                self.spans(line);
342            } else {
343                self.out.keep(line);
344            }
345        }
346    }

348    /// Copies the table starting at the line `start`, if one does, and
349    /// returns the index of the line after it.
350    ///
351    /// A table is a run of lines with a `|` outside code, ending at a blank
352    /// line, a line of attribute lists, or the end. One without a separator
353    /// line gets a header row, for the reader to leave out. A `|` inside
354    /// code is escaped, so CommonMark keeps it in the cell as kramdown does.
355    fn table(&mut self, lines: &[&'a str], start: usize) -> Option<usize> {
356        let end = (start..lines.len())
357            .find(|&j| TableLine::pipes(lines[j], self.math).is_empty())
358            .unwrap_or(lines.len());
359        if end == start {
360            return None;
361        }
362        let after = lines.get(end).map_or("", |line| line.trim());
363        let lists_after = AttributeLists::whole(after).is_some();
364        let rows = &lines[start..end];
365        let separators = rows.iter().filter(|row| TableLine::separator(row)).count();
366        if !(after.is_empty() || lists_after) || separators == rows.len() {
367            return None;
368        }
369        if separators == 0 {
370            let columns = rows
371                .iter()
372                .map(|row| TableLine::cells(row, self.math))
373                .max()
374                .unwrap_or(1);
375            let mut head = "|".to_owned();
376            head.push_str(&" |".repeat(columns));
377            head.push_str("\n|");
378            head.push_str(&"-|".repeat(columns));
379            head.push('\n');
380            self.marks.push(Mark {
381                at: self.out.out.len(),
382                block: true,
383                stands_for: StandIn::Nothing,
384            });
385            self.out.insert(&head);
386        } else if !rows.get(1).is_some_and(|row| TableLine::separator(row)) {
387            // A separator further on, which CommonMark has no way to read.
388            return None;
389        }
390        // A header's separator as CommonMark reads one: a cell for each of
391        // the header's, between `|`.
392        let delimiter = (separators > 0)
393            .then(|| TableLine::delimiter(rows[1], TableLine::cells(rows[0], self.math)))
394            .flatten();
395        for (n, row) in rows.iter().enumerate() {
396            if n == 1
397                && let Some(delimiter) = &delimiter
398            {
399                let content = row.trim_end().len();
400                self.out.replace(content, delimiter);
401                self.out.keep(&row[content..]);
402                continue;
403            }
404            let mut kept = 0;
405            for at in TableLine::code_pipes(row) {
406                self.out.keep(&row[kept..at]);
407                self.out.insert("\\");
408                kept = at;
409            }
410            self.out.keep(&row[kept..]);
411        }
412        Some(end)
413    }

415    /// Copies `before`, then writes a stand-in for `stands_for` in place of
416    /// the next `len` bytes.
417    fn stand_in(&mut self, before: &str, len: usize, block: bool, stands_for: StandIn<'a>) {
418        self.out.keep(before);
419        self.marks.push(Mark {
420            at: self.out.out.len(),
421            block,
422            stands_for,
423        });
424        let text = if block { Self::BLOCK } else { Self::SPAN };
425        self.out.replace(len, text);
426    }

428    /// Replaces the block extension starting on the line `start`, after its
429    /// `lead`, with a stand-in, if it's closed on a line of its own.
430    ///
431    /// Returns the index of the closing line. Its line break is left for
432    /// the caller to copy.
433    fn block(&mut self, lines: &[&'a str], start: usize, lead: usize) -> Option<usize> {
434        let line = lines[start];
435        let content = line[lead..].trim_end();
436        let tag = self
437            .start_tag(content)
438            .filter(|tag| tag.len == content.len())?;
439        if tag.closed {
440            self.stand_in(&line[..lead], content.len(), true, StandIn::Nothing);
441            return Some(start);
442        }
443        let end = (start + 1..lines.len()).find(|&j| Self::closes(lines[j].trim(), tag.name))?;
444        let body = &self.markdown[self.offset(lines[start + 1])..self.offset(lines[end])];
445        let len = self.offset(lines[end]) + lines[end].trim_end().len() - self.offset(line) - lead;
446        let stands_for = Self::giving(&tag, body);
447        self.stand_in(&line[..lead], len, true, stands_for);
448        Some(end)
449    }

451    /// Where `line`, a part of the markdown, starts in it.
452    fn offset(&self, line: &str) -> usize {
453        line.as_ptr() as usize - self.markdown.as_ptr() as usize
454    }

456    /// Copies `line`, replacing each span extension outside code with a
457    /// stand-in and escaping any extension tags left over.
458    fn spans(&mut self, line: &'a str) {
459        let bytes = line.as_bytes();
460        let mut kept = 0;
461        let mut i = 0;
462        while i < line.len() {
463            match bytes[i] {
464                b'\\' => i += 2,
465                b'`' => {
466                    // Past the code span, if it's closed on the line.
467                    let (run, close) = code_span(line, i);
468                    i = close.unwrap_or(i) + run;
469                }
470                b'{' => {
471                    let rest = &line[i..];
472                    if let Some(tag) = self.start_tag(rest) {
473                        let closing = if tag.closed {
474                            Some((tag.len, tag.len))
475                        } else {
476                            Self::find_end(&rest[tag.len..], tag.name)
477                                .map(|(at, len)| (tag.len + at, tag.len + at + len))
478                        };
479                        if let Some((body_end, end)) = closing {
480                            let stands_for = if tag.closed {
481                                StandIn::Nothing
482                            } else {
483                                Self::giving(&tag, &rest[tag.len..body_end])
484                            };
485                            self.stand_in(&line[kept..i], end, false, stands_for);
486                            i += end;
487                            kept = i;
488                            continue;
489                        }
490                    }
491                    if self.start_tag(rest).is_some() || self.end_tag(rest).is_some() {
492                        self.out.keep(&line[kept..i]);
493                        self.out.insert("\\");
494                        kept = i;
495                    }
496                    i += 1;
497                }
498                _ => i += 1,
499            }
500        }
501        self.out.keep(&line[kept..]);
502    }

504    /// What an extension that starts with `tag` and holds `body` stands for.
505    fn giving(tag: &StartTag<'_>, body: &'a str) -> StandIn<'a> {
506        match tag.name {
507            "comment" => StandIn::Comment(body),
508            _ if tag.html => StandIn::Raw(body),
509            _ => StandIn::Nothing,
510        }
511    }

513    /// The start tag at the start of `text`, if it's for an extension that's
514    /// read.
515    fn start_tag<'t>(&self, text: &'t str) -> Option<StartTag<'t>> {
516        let inside = text.strip_prefix("{::")?;
517        let close = inside.find('}')?;
518        let inside = &inside[..close];
519        let name_len = inside
520            .find(|c: char| !(c.is_ascii_alphanumeric() || c == '_'))
521            .unwrap_or(inside.len());
522        let name = &inside[..name_len];
523        let after = &inside[name_len..];
524        if !self.reads(name) || !(after.is_empty() || after.starts_with([' ', '\t', '/'])) {
525            return None;
526        }
527        let closed = after.trim_end().ends_with('/');
528        let html = name != "nomarkdown"
529            || Self::type_of(after).is_none_or(|kind| kind.is_empty() || kind.contains("html"));
530        Some(StartTag {
531            name,
532            len: "{::".len() + close + 1,
533            closed,
534            html,
535        })
536    }

538    /// The length of the end tag at the start of `text`, `{:/name}` or
539    /// `{:/}`, if it's for an extension that's read.
540    fn end_tag(&self, text: &str) -> Option<usize> {
541        let inside = text.strip_prefix("{:/")?;
542        let name = &inside[..inside.find('}')?];
543        (name.is_empty() || self.reads(name)).then(|| "{:/}".len() + name.len())
544    }
545
546    /// Where the first end tag for `name` in `text` is, and its length.
547    fn find_end(text: &str, name: &str) -> Option<(usize, usize)> {
548        text.match_indices("{:/").find_map(|(at, _)| {
549            let inside = &text[at + "{:/".len()..];
550            let tag = &inside[..inside.find('}')?];
551            (tag.is_empty() || tag == name).then(|| (at, "{:/}".len() + tag.len()))
552        })
553    }
554
555    /// Whether `trimmed` is a line of only an end tag for `name`.
556    fn closes(trimmed: &str, name: &str) -> bool {
557        Self::find_end(trimmed, name).is_some_and(|(at, len)| at == 0 && len == trimmed.len())
558    }

560    /// The value of the `type` in the rest of a start tag, if any.
561    fn type_of(rest: &str) -> Option<&str> {
562        let after = &rest[rest.find("type=")? + "type=".len()..];
563        let quote = after.chars().next().filter(|c| matches!(c, '"' | '\''))?;
564        let value = &after[1..];
565        Some(&value[..value.find(quote)?])
566    }
567
568    /// Whether the extension `name` is read.
569    fn reads(&self, name: &str) -> bool {
570        match name {
571            "comment" => self.comments,
572            "nomarkdown" => self.nomarkdown,
573            _ => false,
574        }
575    }

578/// How long the run of backticks at `open` in `line` is, and where the
579/// next run of as many starts, closing the code span it opens, if one
580/// does on the line.
581fn code_span(line: &str, open: usize) -> (usize, Option<usize>) {
582    let run_at = |at: usize| line[at..].len() - line[at..].trim_start_matches('`').len();
583    let run = run_at(open);
584    let mut from = open + run;
585    while let Some(at) = line[from..].find('`').map(|at| from + at) {
586        let len = run_at(at);
587        if len == run {
588            return (run, Some(at));
589        }
590        from = at + len;
591    }
592    (run, None)
593}

609        let at: Vec<_> = marks.iter().map(|mark| (mark.at, mark.block)).collect();

611        assert_eq!(offsets.was(src, &text, "<!---->"), "^");
612        offsets.assert_fit(src, &text);

615        let Marked { text, marks, .. } =
616            StandIns::rewrite(src, Extensions::ENABLE_END_OF_BLOCK_MARKERS);
617        assert!(matches!(text, Cow::Borrowed(_)));
618        assert!(marks.is_empty());
619    }

649    fn stand_ins_rewrite_extensions_closed_and_escape_the_rest() {

651            d {::comment}e{:/} `{::comment}f{:/}` {::nomarkdown/} {:/} {::comment} {:/other}\n";
652        let Marked {
653            text,
654            offsets,
655            marks,
656            ..
657        } = StandIns::rewrite(
658            src,
659            Extensions::ENABLE_COMMENTS | Extensions::ENABLE_NOMARKDOWN,
660        );
661        assert_eq!(
662            text,
663            "<!---->\n<!---->\nd <kd:x> `{::comment}f{:/}` <kd:x> \\{:/} \\{::comment} {:/other}\n"

671                StandIn::Comment("e"),

674        );
